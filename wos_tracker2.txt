        function executeLogin() {
            const user = document.getElementById('auth_user').value.trim();
            const pass = document.getElementById('auth_pass').value.trim();

            if(!user || !pass) return alert("Fill out all identity fields.");

            // Master Administrator Overrides
            if (pass === "102953") {
                currentUser = { username: user || 'MasterAdmin', role: 'MasterAdmin', linkedId: null };
                configureUIForRole();
                closeModal('loginModal');
                return alert("Access Granted: Root Authority Cleared.");
            }

            // Sync Verification Strategy
            if (useFirebase) {
                db.collection("users").doc(user).get().then(doc => {
                    if (doc.exists && doc.data().passkey === pass) {
                        currentUser = doc.data();
                        configureUIForRole();
                        closeModal('loginModal');
                        alert(`Welcome back, ${user}. Role: ${currentUser.role}`);
                    } else {
                        alert("Invalid identity alignment parameters.");
                    }
                });
            } else {
                if (systemUsers[user] && systemUsers[user].passkey === pass) {
                    currentUser = systemUsers[user];
                    configureUIForRole();
                    closeModal('loginModal');
                    alert(`Welcome back, ${user}. Role: ${currentUser.role}`);
                } else {
                    alert("Invalid identity credentials.");
                }
            }
        }

        function executeRegister() {
            const user = document.getElementById('auth_user').value.trim();
            const pass = document.getElementById('auth_pass').value.trim();

            if(!user || !pass) return alert("Define username and passkey components clearly.");
            if(pass === "102953") return alert("Reserved authority cryptographic key.");

            const newUserObj = { username: user, passkey: pass, role: 'Player', linkedId: null };

            if (useFirebase) {
                db.collection("users").doc(user).get().then(doc => {
                    if (doc.exists) {
                        alert("Identity identity profile variant signature already exists.");
                    } else {
                        db.collection("users").doc(user).set(newUserObj).then(() => {
                            systemUsers[user] = newUserObj;
                            currentUser = newUserObj;
                            configureUIForRole();
                            closeModal('loginModal');
                            alert("Account established securely via fully anonymous allocation frameworks.");
                        });
                    }
                });
            } else {
                if (systemUsers[user]) return alert("Identity already configured locally.");
                systemUsers[user] = newUserObj;
                currentUser = newUserObj;
                configureUIForRole();
                closeModal('loginModal');
                alert("Account established locally.");
            }
        }

        // --- PROFILE SUBMISSION AND STORAGE ENGINE DATA HANDLERS ---
        function submitPlayerProfile() {
            const id = document.getElementById('p_id').value.trim();
            const name = document.getElementById('p_name').value.trim();
            const alliance = document.getElementById('p_alliance').value.trim().toUpperCase();
            const rank = document.getElementById('p_rank').value;
            const power = parseInt(document.getElementById('p_power').value) || 0;

            if (!id || !name || !alliance) return alert("Required parameters: ID Number, Name, Alliance TAG.");

            // Extract multi-select checkbox structures
            const tiers = [];
            if(document.getElementById('t_fc7').checked) tiers.push('FC7');
            if(document.getElementById('t_fc8').checked) tiers.push('FC8');
            if(document.getElementById('t_fc9').checked) tiers.push('FC9');
            if(document.getElementById('t_fc10').checked) tiers.push('FC10');

            const helios = [];
            if(document.getElementById('h_inf').checked) helios.push('Helios Infantry');
            if(document.getElementById('h_lan').checked) helios.push('Helios Lancers');
            if(document.getElementById('h_mar').checked) helios.push('Helios Marksman');

            // Extract Hero selections
            const goldHeroStars = {};
            for (const heroes of Object.values(GOLD_HERO_BY_GEN)) {
                heroes.forEach(hero => {
                    const elId = `hero_star_${hero.replace(/\s+/g, '_')}`;
                    goldHeroStars[hero] = parseInt(document.getElementById(elId).value) || 0;
                });
            }

            const profilePayload = {
                id, name, alliance, rank, power,
                tiers, helios, goldHeroStars,
                timestamp: Date.now(),
                approved: (currentUser.role === 'MasterAdmin' || currentUser.role === 'R4' || currentUser.role === 'R5')
            };

            // Gate validations check linked updates restrictions mapping
            if (currentUser.role === 'Player' && currentUser.linkedId && currentUser.linkedId !== id) {
                return alert("Access Denied: Your account is locked to a separate Player ID.");
            }

            if (useFirebase) {
                const targetColl = profilePayload.approved ? "players" : "submissions";
                db.collection(targetColl).doc(id).set(profilePayload).then(() => {
                    if (currentUser.role === 'Player' && !currentUser.linkedId) {
                        currentUser.linkedId = id;
                        db.collection("users").doc(currentUser.username).update({ linkedId: id });
                    }
                    alert(profilePayload.approved ? "Profile updated live instantly." : "Submission queued for administrative R4/R5 clearance reviews.");
                    clearPlayerForm();
                    loadInitialDatabase();
                });
            } else {
                if (profilePayload.approved) {
                    playersDB[id] = profilePayload;
                    if (currentUser.role === 'Player' && !currentUser.linkedId) currentUser.linkedId = id;
                    alert("Profile synchronized to core ledger cache.");
                } else {
                    if (!window.localSubmissionsQueue) window.localSubmissionsQueue = [];
                    window.localSubmissionsQueue.push(profilePayload);
                    alert("Submission held locally inside browser cache queue for admin verification testing.");
                }
                clearPlayerForm();
                renderRosterGrid();
                renderAdminPanelUserMatrix();
            }
        }

        function loadInitialDatabase() {

                    runQuickSearch();

        function renderRosterGrid() {
            const tbody = document.getElementById('rosterTableBody');
            tbody.innerHTML = '';
            const searchVal = document.getElementById('rosterSearch').value.toLowerCase().trim();

            let playersArray = Object.values(playersDB);

            // Filter logic matches
            if (searchVal) {
                playersArray = playersArray.filter(p => p.name.toLowerCase().includes(searchVal) || p.id.includes(searchVal));
            }

            // Sorting logistics arrays metrics mapping
            playersArray.sort((a, b) => {
                let fieldA = a[rosterSortKey];
                let fieldB = b[rosterSortKey];
                
                if (typeof fieldA === 'string') {
                    return rosterSortAsc ? fieldA.localeCompare(fieldB) : fieldB.localeCompare(fieldA);
                } else {
                    return rosterSortAsc ? fieldA - fieldB : fieldB - fieldA;
                }
            });

            const isAdmin = ['MasterAdmin', 'R4', 'R5'].includes(currentUser.role);

            playersArray.forEach(p => {
                let maxedGolds = [];
                if (p.goldHeroStars) {
                    for (const [hero, stars] of Object.entries(p.goldHeroStars)) {
                        if (stars === 5) maxedGolds.push(hero);
                    }
                }

                const tr = document.createElement('tr');
                tr.innerHTML = `
                    <td><code>${p.id}</code></td>
                    <td><strong>${p.name}</strong></td>
                    <td><span class="badge badge-blue">${p.alliance}</span></td>
                    <td>${p.rank}</td>
                    <td>${(p.power / 1000000).toFixed(1)}M</td>
                    <td><small>${p.tiers.join(', ') || 'None'} <br> ${p.helios.join(', ') || 'No Helios'}</small></td>
                    <td><small>${maxedGolds.join(', ') || 'None Maxed'}</small></td>
                    ${isAdmin ? `<td>
                        <button class="btn btn-warning" style="padding:4px 8px; font-size:12px;" onclick="editPlayerInline('${p.id}')">Edit</button>
                        <button class="btn btn-danger" style="padding:4px 8px; font-size:12px;" onclick="deletePlayerInline('${p.id}')">Remove</button>
                    </td>` : ''}
                `;
                tbody.appendChild(tr);
            });
        }

        function sortRoster(key) {
            if (rosterSortKey === key) {
                rosterSortAsc = !rosterSortAsc;
            } else {
                rosterSortKey = key;
                rosterSortAsc = true;
            }
            renderRosterGrid();
        }

        // --- QUICK SEARCH COMBAT SQUAD FILTERS ENGINE ---
        function runQuickSearch() {
            const tbody = document.getElementById('searchResultsBody');
            tbody.innerHTML = '';

            const reqTiers = Array.from(document.querySelectorAll('.filter-tier:checked')).map(el => el.value);
            const reqHelios = Array.from(document.querySelectorAll('.filter-helios:checked')).map(el => el.value);
            
            const selectHeroEl = document.getElementById('filterHeroSelect');
            const reqHeroes = Array.from(selectHeroEl.selectedOptions).map(opt => opt.value);

            let matches = Object.values(playersDB);

            // Filter process checks logic
            if (reqTiers.length > 0) {
                matches = matches.filter(p => reqTiers.some(t => p.tiers.includes(t)));
            }
            if (reqHelios.length > 0) {
                matches = matches.filter(p => reqHelios.every(h => p.helios.includes(h)));
            }
            if (reqHeroes.length > 0) {
                matches = matches.filter(p => p.goldHeroStars && reqHeroes.every(h => p.goldHeroStars[h] === 5));
            }

            if (matches.length === 0) {
                tbody.innerHTML = `<tr><td colspan="6" style="text-align:center; color:var(--text-muted);">No deployment configurations match your exact filtering criteria.</td></tr>`;
                return;
            }

            matches.forEach(p => {
                const tr = document.createElement('tr');
                tr.innerHTML = `
                    <td><code>${p.id}</code></td>
                    <td><strong>${p.name}</strong></td>
                    <td><span class="badge badge-blue">${p.alliance}</span></td>
                    <td>${p.rank}</td>
                    <td>${(p.power / 1000000).toFixed(1)}M</td>
                    <td><span style="color:var(--success);">✔ Valid Joiner Profile</span></td>
                `;
                tbody.appendChild(tr);
            });
        }
